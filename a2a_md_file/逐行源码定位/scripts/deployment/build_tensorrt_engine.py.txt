00056 | # STRONGLY_TYPED precision sanity check: TRT 10+ STRONGLY_TYPED reads
00057 | # precision from the ONNX tensor types and ignores --precision builder
00058 | # flags. Catch the silent mismatch (user asks fp16, ONNX is bf16, engine
00059 | # silently builds bf16) before burning build time. Indirected through
00060 | # dtype *names* so the helper can be unit-tested without TensorRT.
00061 | _PRECISION_TO_TRT_DTYPE_NAME: dict[str, str] = {
00062 |     "bf16": "BF16",
00063 |     "fp16": "HALF",
00064 |     "fp32": "FLOAT",
00065 |     "fp8": "FP8",

00069 | def _check_strongly_typed_precision_match(
00070 |     network_dtype_names: set[str], requested_precision: str
00071 | ) -> None:
00072 |     """Raise if --precision cannot be honored by this STRONGLY_TYPED network."""
00073 |     expected = _PRECISION_TO_TRT_DTYPE_NAME.get(requested_precision)
00074 |     if expected is None:
00075 |         raise ValueError(
00076 |             f"Unknown precision: {requested_precision!r}. "
00077 |             f"Expected one of {sorted(_PRECISION_TO_TRT_DTYPE_NAME)}."
00078 |         )
00079 |     if expected not in network_dtype_names:
00080 |         raise ValueError(
00081 |             f"--precision={requested_precision} cannot be honored by this ONNX. "
00082 |             f"STRONGLY_TYPED (TRT 10+) reads precision from ONNX tensor types "
00083 |             f"and ignores builder flags. Network has tensor dtypes "
00084 |             f"{sorted(network_dtype_names)}; none of them are {expected}. "
00085 |             f"Either re-export the ONNX with the requested precision, or "
00086 |             f"pass --precision matching the existing ONNX dtypes."
00087 |         )
00088 |     # When fp32 is requested, the network must not contain any reduced-precision
00089 |     # tensors. STRONGLY_TYPED won't promote BF16/FP16/FP8 to FLOAT, so a mixed
00090 |     # BF16+FLOAT network silently runs at BF16 for those tensors despite the
00091 |     # caller asking for fp32.
00092 |     if requested_precision == "fp32":
00093 |         reduced = {"BF16", "HALF", "FP8"} & network_dtype_names
00094 |         if reduced:
00095 |             raise ValueError(
00096 |                 f"--precision=fp32 cannot be honored: network also contains "
00097 |                 f"reduced-precision tensors {sorted(reduced)}. STRONGLY_TYPED "
00098 |                 f"won't promote them to FLOAT, so the engine would silently "
00099 |                 f"run mixed precision. Re-export the ONNX as pure FP32, or "
00100 |                 f"pass --precision matching the dominant reduced dtype."
00101 |             )

00104 | def _precision_from_onnx_path(onnx_path: str, default: str) -> str:
00105 |     """Return the precision tag suffixed in the ONNX filename (e.g.
00106 |     ``vit_fp32.onnx`` → ``"fp32"``), else ``default``. Used so the
00107 |     full-pipeline build mirrors the export's per-component dtype instead
00108 |     of forwarding the pipeline-wide ``--precision`` to a mismatched ONNX.
00109 |     """
00110 |     stem = os.path.splitext(os.path.basename(onnx_path))[0]
00111 |     for tag in _PRECISION_TO_TRT_DTYPE_NAME:

00113 |             return tag
00114 |     return default

00122 | def derive_shapes_from_onnx(onnx_path, max_batch=8):
00123 |     """Read an ONNX model and derive min/opt/max shape profiles.
00124 | 
00125 |     For each input:
00126 |     - Fixed dimensions (concrete values) are kept as-is across min/opt/max.
00127 |     - Dynamic batch dimension: min=1, opt=1, max=max_batch.
00128 |     - Dynamic sequence dimensions: min=1, opt=concrete_value, max=2*concrete_value.
00129 |       (concrete_value comes from the ONNX model's shape hints)
00130 | 
00131 |     Returns (min_shapes, opt_shapes, max_shapes) dicts.
00132 |     """
00133 |     model = onnx.load(onnx_path, load_external_data=False)
00134 | 
00135 |     min_shapes, opt_shapes, max_shapes = {}, {}, {}
00136 | 
00137 |     for inp in model.graph.input:
00138 |         name = inp.name
00139 |         dims = inp.type.tensor_type.shape.dim
00140 | 
00141 |         min_shape, opt_shape, max_shape = [], [], []
00142 |         for i, d in enumerate(dims):
00143 |             if d.dim_value > 0:
00144 |                 # Fixed dimension — use as-is
00145 |                 min_shape.append(d.dim_value)
00146 |                 opt_shape.append(d.dim_value)
00147 |                 max_shape.append(d.dim_value)
00148 |             else:
00149 |                 # Dynamic dimension
00150 |                 if i == 0:
00151 |                     # Batch dimension
00152 |                     min_shape.append(1)
00153 |                     opt_shape.append(1)
00154 |                     max_shape.append(max_batch)
00155 |                 else:
00156 |                     # Sequence/spatial dimension — use generous range
00157 |                     # We don't know the "typical" value from ONNX alone,
00158 |                     # so use 1 / 1 / large_max. The builder will optimize for opt.
00159 |                     min_shape.append(1)
00160 |                     opt_shape.append(1)
00161 |                     max_shape.append(512)
00162 | 
00163 |         min_shapes[name] = tuple(min_shape)
00164 |         opt_shapes[name] = tuple(opt_shape)
00165 |         max_shapes[name] = tuple(max_shape)
00166 | 
00167 |     return min_shapes, opt_shapes, max_shapes

00170 | def derive_shapes_with_hint(onnx_path, opt_seq_lens=None, max_batch=8):
00171 |     """Derive shapes from ONNX, with optional sequence length hints.
00172 | 
00173 |     Args:
00174 |         onnx_path: Path to ONNX model
00175 |         opt_seq_lens: Dict mapping dynamic dim names to optimal sequence lengths.
00176 |                       e.g. {"sa_seq_len": 51, "vl_seq_len": 280, "sequence_length": 280}
00177 |         max_batch: Maximum batch size
00178 |     """
00179 |     model = onnx.load(onnx_path, load_external_data=False)
00180 |     opt_seq_lens = opt_seq_lens or {}
00181 | 
00182 |     min_shapes, opt_shapes, max_shapes = {}, {}, {}
00183 | 
00184 |     for inp in model.graph.input:
00185 |         name = inp.name
00186 |         dims = inp.type.tensor_type.shape.dim
00187 | 
00188 |         min_shape, opt_shape, max_shape = [], [], []
00189 |         for i, d in enumerate(dims):
00190 |             if d.dim_value > 0:
00191 |                 # Fixed dimension
00192 |                 min_shape.append(d.dim_value)
00193 |                 opt_shape.append(d.dim_value)
00194 |                 max_shape.append(d.dim_value)
00195 |             else:
00196 |                 dim_name = d.dim_param if d.dim_param else f"dim_{i}"
00197 |                 if dim_name in {"batch", "batch_size"}:
00198 |                     # Batch dimension (at any index)
00199 |                     min_shape.append(1)
00200 |                     opt_shape.append(1)
00201 |                     max_shape.append(max_batch)
00202 |                 elif dim_name in opt_seq_lens:
00203 |                     # Named dynamic dim with a hint
00204 |                     opt_val = opt_seq_lens[dim_name]
00205 |                     min_shape.append(1)
00206 |                     opt_shape.append(opt_val)
00207 |                     max_shape.append(max(opt_val * 2, opt_val + 64))
00208 |                 else:
00209 |                     # Unknown dynamic dim — use wide range
00210 |                     min_shape.append(1)
00211 |                     opt_shape.append(256)
00212 |                     max_shape.append(512)
00213 | 
00214 |         min_shapes[name] = tuple(min_shape)
00215 |         opt_shapes[name] = tuple(opt_shape)
00216 |         max_shapes[name] = tuple(max_shape)
00217 | 
00218 |     return min_shapes, opt_shapes, max_shapes

00390 | def build_full_pipeline(
