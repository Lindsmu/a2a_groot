00091 | def _rec_to_dtype(x, dtype):
00092 |     """Recursively convert all floating point tensors to the given dtype."""
00093 |     if isinstance(x, torch.Tensor) and torch.is_floating_point(x):
00094 |         return x.to(dtype=dtype)
00095 |     elif isinstance(x, dict) or hasattr(x, "items"):
00096 |         return {k: _rec_to_dtype(v, dtype) for k, v in x.items()}
00097 |     elif isinstance(x, list):
00098 |         return [_rec_to_dtype(v, dtype) for v in x]
00099 |     else:
00100 |         return x

00103 | def prepare_model_inputs(policy, observation, return_states=False):
00104 |     """
00105 |     Prepare inputs for the model, mimicking what happens inside _get_action.
00106 |     Returns collated_inputs that can be passed to model.get_action()

00108 |     Args:
00109 |         policy: The Gr00tPolicy instance
00110 |         observation: Dict with "video", "state", "language" keys
00111 |         return_states: If True, also return the states list (for action denormalization)
00112 | 
00113 |     Returns:
00114 |         collated_inputs if return_states=False, else (collated_inputs, states)
00115 |     """
00116 |     unbatched_obs = []
00117 |     batch_size = observation["video"][list(observation["video"].keys())[0]].shape[0]
00118 |     for i in range(batch_size):
00119 |         unbatched_value = {
00120 |             "video": {k: v[i] for k, v in observation["video"].items()},
00121 |             "state": {k: v[i] for k, v in observation["state"].items()},
00122 |             "language": {k: v[i] for k, v in observation["language"].items()},
00123 |         }
00124 |         unbatched_obs.append(unbatched_value)
00125 | 
00126 |     processed_inputs = []
00127 |     states = []
00128 |     for obs in unbatched_obs:
00129 |         vla_step_data = VLAStepData(
00130 |             images=obs["video"],
00131 |             states=obs["state"],
00132 |             actions={},
00133 |             text=obs["language"][policy.language_key][0],
00134 |             embodiment=policy.embodiment_tag,
00135 |         )
00136 |         states.append(vla_step_data.states)

00138 |         processed_inputs.append(policy.processor(messages))
00139 | 
00140 |     collated_inputs = policy.collate_fn(processed_inputs)
00141 |     collated_inputs = collated_inputs["inputs"]

00179 |     Benchmark data processing separately with proper warmup.

00188 |     If observation is a list (trajectory), cycles through observations during benchmarking.
00189 |     """
00190 |     import gc
00191 | 
00192 |     # Handle both single observation and trajectory (list of observations)
00193 |     if isinstance(observation, list):
00194 |         observations = observation
00195 |     else:
00196 |         observations = [observation]
00197 | 
00198 |     num_obs = len(observations)
00199 | 
00200 |     # Force GC before warmup to reduce variance
00201 |     gc.collect()
00202 | 
00203 |     # Warmup - helps with CPU caching and JIT for consistent benchmarks
00204 |     # For trajectory mode, warmup benefit is reduced since each observation is different
00205 |     if warmup > 0:
00206 |         for i in range(warmup):
00207 |             obs = observations[i % num_obs]
00208 |             _ = prepare_model_inputs(policy, obs)

00210 |         gc.collect()

00213 |     times = []

00215 |         obs = observations[i % num_obs]
