00126 | def verify_onnx_with_ort(
00127 |     onnx_path: str,
00128 |     pytorch_module: torch.nn.Module,
00129 |     sample_inputs: dict[str, torch.Tensor],
00130 |     output_names: list[str],
00131 |     label: str = "model",
00132 | ) -> dict[str, float]:
00133 |     """Run ONNX Runtime inference and compare against PyTorch output.
00134 | 
00135 |     Returns dict of {output_name: cosine_similarity}.
00136 |     Requires onnxruntime-gpu; skips gracefully if not installed.
00137 |     """
00138 |     try:
00139 |         import onnxruntime as ort
00140 |     except ImportError:
00141 |         logger.warning("  onnxruntime not installed — skipping ORT verification")
00142 |         return {}
00143 | 
00144 |     logger.info(f"  ORT verification for {label}...")
00145 | 
00146 |     # Run PyTorch
00147 |     with torch.inference_mode():
00148 |         pt_inputs = tuple(sample_inputs[name] for name in sample_inputs)
00149 |         pt_outputs = pytorch_module(*pt_inputs)
00150 |         if not isinstance(pt_outputs, (tuple, list)):
00151 |             pt_outputs = (pt_outputs,)
00152 | 
00153 |     # Run ONNX Runtime
00154 |     providers = ["CUDAExecutionProvider", "CPUExecutionProvider"]
00155 |     sess = ort.InferenceSession(onnx_path, providers=providers)
00156 |     ort_inputs = {name: t.cpu().numpy() for name, t in sample_inputs.items()}
00157 |     ort_outputs = sess.run(output_names, ort_inputs)
00158 | 
00159 |     # Compare
00160 |     results = {}
00161 |     for i, name in enumerate(output_names):
00162 |         pt_flat = pt_outputs[i].float().flatten().cpu()
00163 |         ort_flat = torch.tensor(ort_outputs[i]).float().flatten()
00164 |         cosine = torch.nn.functional.cosine_similarity(
00165 |             pt_flat.unsqueeze(0), ort_flat.unsqueeze(0)
00166 |         ).item()
00167 |         results[name] = cosine
00168 |         logger.info(f"    {name}: ORT vs PyTorch cosine = {cosine:.6f}")
00169 | 
00170 |     return results

00302 | def _apply_rotary_real(x, cos, sin):
00303 |     """Apply rotary position embeddings using only real-valued ops (no complex).
00304 | 
00305 |     Uses float32 internally to match transformers' apply_rotary_pos_emb_vision
00306 |     precision, then casts back to the original dtype.
00307 | 
00308 |     Args:
00309 |         x: [seq, heads, head_dim]
00310 |         cos, sin: [seq, head_dim]
00311 |     Returns:
00312 |         [seq, heads, head_dim]
00313 |     """
00314 |     orig_dtype = x.dtype
00315 |     x = x.float()
00316 |     cos = cos.float().unsqueeze(1)  # [seq, 1, head_dim]
00317 |     sin = sin.float().unsqueeze(1)
00318 |     half = x.shape[-1] // 2
00319 |     x1 = x[..., :half]
00320 |     x2 = x[..., half:]
00321 |     rotated = torch.cat((-x2, x1), dim=-1)
00322 |     return (x * cos + rotated * sin).to(orig_dtype)

00325 | def _make_onnx_vision_attention_forward(attn_module, chunk_sizes=None):
00326 |     """Create an ONNX-exportable attention forward for a single VisionAttention.
00327 | 
00328 |     Three key changes from the original:
00329 |     1. Replaces cu_seqlens-based splitting with static chunk splitting
00330 |        (each image's patches attend only within their own chunk)
00331 |     2. Replaces apply_rotary_pos_emb_vision (uses complex numbers) with
00332 |        real-valued rotate_half implementation
00333 |     3. Casts to float32 before softmax for TRT accuracy
00334 | 
00335 |     Args:
00336 |         attn_module: The VisionAttention module to wrap
00337 |         chunk_sizes: List of ints, number of patches per image.
00338 |             e.g. [256, 256] for 2 images of 256 patches each.
00339 |             If None or single chunk, does full-sequence attention.
00340 |     """
00341 | 
00342 |     def forward(
00343 |         hidden_states, cu_seqlens=None, rotary_pos_emb=None, position_embeddings=None, **kwargs
00344 |     ):
00345 |         seq_length = hidden_states.shape[0]
00346 |         qkv = attn_module.qkv(hidden_states)
00347 |         qkv = qkv.reshape(seq_length, 3, attn_module.num_heads, -1)
00348 |         qkv = qkv.permute(1, 0, 2, 3)
00349 |         q, k, v = qkv.unbind(0)
00350 |         # q, k, v: [seq_length, num_heads, head_dim]
00351 | 
00352 |         cos, sin = position_embeddings
00353 |         q = _apply_rotary_real(q, cos, sin)
00354 |         k = _apply_rotary_real(k, cos, sin)
00355 | 
00356 |         # Split by image chunks (mirrors cu_seqlens-based splitting in original)
00357 |         # Each image's patches attend only within their own chunk.
00358 |         if chunk_sizes is not None and len(chunk_sizes) > 1:
00359 |             q_chunks = torch.split(q, chunk_sizes, dim=0)
00360 |             k_chunks = torch.split(k, chunk_sizes, dim=0)
00361 |             v_chunks = torch.split(v, chunk_sizes, dim=0)
00362 | 
00363 |             attn_outputs = []
00364 |             for q_c, k_c, v_c in zip(q_chunks, k_chunks, v_chunks):
00365 |                 # q_c, k_c, v_c: [chunk_seq, num_heads, head_dim]
00366 |                 q_c = q_c.transpose(0, 1)  # [num_heads, chunk_seq, head_dim]
00367 |                 k_c = k_c.transpose(0, 1)
00368 |                 v_c = v_c.transpose(0, 1)
00369 | 
00370 |                 w = torch.matmul(q_c, k_c.transpose(-2, -1)) * attn_module.scaling
00371 |                 w = w.to(torch.float32)
00372 |                 w = F.softmax(w, dim=-1)
00373 |                 w = w.to(v_c.dtype)
00374 |                 out = torch.matmul(w, v_c)  # [num_heads, chunk_seq, head_dim]
00375 |                 attn_outputs.append(out.transpose(0, 1))  # [chunk_seq, num_heads, head_dim]
00376 | 
00377 |             attn_output = torch.cat(attn_outputs, dim=0)  # [seq, num_heads, head_dim]
00378 |         else:
00379 |             # Single image: full-sequence attention
00380 |             q = q.transpose(0, 1)  # [num_heads, seq, head_dim]
00381 |             k = k.transpose(0, 1)
00382 |             v = v.transpose(0, 1)
00383 | 
00384 |             attn_weights = torch.matmul(q, k.transpose(-2, -1)) * attn_module.scaling
00385 |             attn_weights = attn_weights.to(torch.float32)
00386 |             attn_weights = F.softmax(attn_weights, dim=-1)
00387 |             attn_weights = attn_weights.to(v.dtype)
00388 |             attn_output = torch.matmul(attn_weights, v)
00389 |             attn_output = attn_output.transpose(0, 1)  # [seq, num_heads, head_dim]
00390 | 
00391 |         # [seq, num_heads, head_dim] → [seq, num_heads * head_dim]
00392 |         attn_output = attn_output.reshape(seq_length, -1).contiguous()
00393 |         attn_output = attn_module.proj(attn_output)
00394 |         return attn_output
00395 | 
00396 |     return forward

00427 | class Qwen3VisionForExport(torch.nn.Module):
00428 |     """ONNX-exportable wrapper for Qwen3-VL Vision Model.
00429 | 
00430 |     Pre-computes position embeddings and rotary embeddings for a fixed grid_thw
00431 |     to avoid ComplexDouble operations that ONNX cannot handle. Replaces the
00432 |     dynamic VisionModel.forward with a traceable version.
00433 | 
00434 |     Architecture: patch_embed → add pos_embed → blocks(attn+ffn) → deepstack → merger
00435 |     """
00436 | 
00437 |     def __init__(self, vision_model, grid_thw: torch.Tensor):
00438 |         super().__init__()
00439 |         self.patch_embed = vision_model.patch_embed
00440 |         self.blocks = vision_model.blocks
00441 |         self.merger = vision_model.merger
00442 |         self.deepstack_visual_indexes = vision_model.deepstack_visual_indexes
00443 |         self.deepstack_merger_list = vision_model.deepstack_merger_list
00444 | 
00445 |         # Pre-compute position embeddings (avoids grid_thw-dependent Python loops
00446 |         # and ComplexDouble operations in rotary embedding computation)
00447 |         with torch.no_grad():
00448 |             pos_embeds = vision_model.fast_pos_embed_interpolate(grid_thw)
00449 |             rotary = vision_model.rot_pos_emb(grid_thw)
00450 |             emb = torch.cat((rotary, rotary), dim=-1)
00451 | 
00452 |         self.register_buffer("_pos_embeds", pos_embeds.clone().detach().contiguous())
00453 |         self.register_buffer("_rot_cos", emb.cos().clone().detach().contiguous())
00454 |         self.register_buffer("_rot_sin", emb.sin().clone().detach().contiguous())
00455 | 
00456 |     def forward(self, pixel_values):
00457 |         hidden_states = self.patch_embed(pixel_values)
00458 |         hidden_states = hidden_states + self._pos_embeds
00459 | 
00460 |         position_embeddings = (self._rot_cos, self._rot_sin)
00461 | 
00462 |         deepstack_features = []
00463 |         for layer_num, blk in enumerate(self.blocks):
00464 |             hidden_states = blk(
00465 |                 hidden_states,
00466 |                 cu_seqlens=None,  # not used by patched attention
00467 |                 position_embeddings=position_embeddings,
00468 |             )
00469 |             if layer_num in self.deepstack_visual_indexes:
00470 |                 idx = self.deepstack_visual_indexes.index(layer_num)
00471 |                 deepstack_features.append(self.deepstack_merger_list[idx](hidden_states))
00472 | 
00473 |         image_embeds = self.merger(hidden_states)
00474 | 
00475 |         if deepstack_features:
00476 |             deepstack = torch.stack(deepstack_features)  # [num_layers, N, D]
00477 |         else:
00478 |             deepstack = image_embeds.new_zeros(1, 1, 1)
00479 | 
00480 |         return image_embeds, deepstack

00488 | def _restore_vision_rotary_inv_freq_fp32(vision, vision_config):
00489 |     """Re-derive the ViT RoPE ``inv_freq`` in fp32 right before baking it into the engine.
00490 | 
00491 |     ``Gr00tPolicy`` loads the whole model in bf16 (``gr00t_policy.py``), which rounds
