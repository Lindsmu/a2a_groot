00078 | def _assert_supports_trt_padding_strip(attention_mask: torch.Tensor) -> None:
00079 |     """Hard-fail if the TRT padding-strip path would silently corrupt B>1 batches.
00080 | 
00081 |     transformers 4.57+ strips padding tokens before calling ``language_model``.
00082 |     The TRT forward path replicates that with ``valid_mask = attention_mask[0] == 1``
00083 |     so engine inputs match export-time captured shapes. With B=1 (or B>1 where
00084 |     every sample has the same mask, e.g. tiled benchmarking via
00085 |     ``verify_n1d7_trt.py --batch_size N``) this is correct. With B>1 *and*
00086 |     heterogeneous masks (different valid lengths per sample), samples 1..B-1
00087 |     would be silently mis-stripped using sample 0's padding template, producing
00088 |     wrong inputs to the LLM TRT engine — no error, no warning, just degraded
00089 |     output that's caught only via downstream eval drift.
00090 | 
00091 |     Raise rather than warn so the caller can't miss it in the multi-rank log
00092 |     torrent. To support heterogeneous B>1, switch to the all-batch
00093 |     common-padding mask ``(attention_mask == 1).all(dim=0)`` (only strips
00094 |     columns padded across *all* samples) — that is a semantic change requiring
00095 |     modeling sign-off, so we hard-fail here for now.
00096 | 
00097 |     Note on the build-time batch coupling: the ONNX export pipeline
00098 |     (``scripts/deployment/export_onnx_n1d7.py``) bakes the batch dim as a
00099 |     static shape — only ``seq_len`` is registered in ``dynamic_axes`` —
00100 |     so a TRT engine built at ``--batch-size N`` only accepts B=N at
00101 |     runtime. Lifting heterogeneous B>1 inference therefore requires both
00102 |     rebuilding the engine at that batch size *and* relaxing this
00103 |     assertion. Production today is B=1; this check covers the runtime
00104 |     half of that contract.
00105 |     """
00106 |     if attention_mask.shape[0] <= 1:
00107 |         return
00108 |     if not (attention_mask == attention_mask[0:1]).all():
00109 |         raise ValueError(
00110 |             "TRT backbone padding-strip requires B=1 or homogeneous attention "
00111 |             f"masks across the batch (got B={attention_mask.shape[0]} with "
00112 |             "heterogeneous masks). attention_mask[0] would silently mis-strip "
00113 |             "samples 1..B-1 using sample 0's padding template, producing wrong "
00114 |             "inputs to the LLM TRT engine. Switch to "
00115 |             "(attention_mask == 1).all(dim=0) to only strip columns padded "
00116 |             "across all samples — but that is a semantic change requiring "
00117 |             "modeling sign-off."
00118 |         )

00121 | def _resolve_vit_engine_path(trt_engine_path: str) -> str:
00122 |     """Locate the ViT engine, tolerating the legacy filename.
00123 | 
00124 |     Older builds emitted ``vit_bf16.engine`` even when the source ONNX
00125 |     was FP32 — a name that obscured the actual engine precision. New
00126 |     builds emit ``vit.engine`` (precision-neutral; the real dtype lives
00127 |     in ``export_metadata.json`` and the ONNX tensor types). Existing
00128 |     engine directories with the legacy name are still accepted with a
00129 |     warning emitted on each call (no per-process suppression: the
00130 |     warning content is path-dependent and call sites are few), to
00131 |     encourage migration to the precision-neutral name. When no engine
00132 |     is present we return the canonical name so the caller can put it
00133 |     in any "not found" error.
00134 |     """
00135 |     new = os.path.join(trt_engine_path, "vit.engine")
00136 |     legacy = os.path.join(trt_engine_path, "vit_bf16.engine")
00137 |     if os.path.exists(new):
00138 |         return new
00139 |     if os.path.exists(legacy):
00140 |         logger.warning(
00141 |             f"Loading ViT engine from legacy path {legacy}; new builds "
00142 |             f"emit vit.engine. Rebuild the pipeline to drop the misleading "
00143 |             f"precision tag from the filename."
00144 |         )
00145 |         return legacy
00146 |     return new

00646 | def close_tensorrt_engines(policy):
00647 |     """Release every TRT ``Engine`` attached to ``policy`` (best-effort).
00648 | 
00649 |     setup_tensorrt_engines stashes engines as plain attributes on the backbone
00650 |     and action head (which subset depends on the mode), so walk the model's
00651 |     submodules and close each. Needed because the sim-eval entrypoint hard-exits
00652 |     via os._exit, which skips ``Engine.__del__`` and would otherwise leak GPU
00653 |     memory into the next eval shard.
00654 |     """
00655 |     # The sim wrapper exposes the inner Gr00tPolicy as ``.policy``; a remote
00656 |     # PolicyClient has no model and no engines to close.
00657 |     model = getattr(getattr(policy, "policy", policy), "model", None)
00658 |     if model is None:
00659 |         return
00660 |     closed = set()
00661 |     for module in model.modules():
00662 |         # Snapshot the attribute values: close() runs C++ destructors, so avoid
00663 |         # iterating a live __dict__ view.
00664 |         for value in list(vars(module).values()):
00665 |             if isinstance(value, Engine) and id(value) not in closed:
00666 |                 closed.add(id(value))
00667 |                 try:
00668 |                     value.close()
00669 |                 except Exception as e:
00670 |                     print(f"Failed to close TRT engine: {e}")
