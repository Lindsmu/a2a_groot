00070 | def _run_a2a_engine(engine, tensors):
00071 |     prepared = {}
00072 |     for name in A2A_ENGINE_INPUT_NAMES:
00073 |         value = tensors[name]
00074 |         expected_dtype = engine.dtype_of(name)
00075 |         value = value.to(dtype=expected_dtype).contiguous()
00076 |         engine.set_runtime_tensor_shape(name, tuple(value.shape))
00077 |         prepared[name] = value
00078 |     return engine(**prepared)
