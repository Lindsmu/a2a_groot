00060 | def _backbone(config, batch_size=2):
00061 |     attention = torch.ones(batch_size, 6, dtype=torch.long)
00062 |     image = torch.zeros(batch_size, 6, dtype=torch.bool)
00063 |     image[:, :3] = True
00064 |     return BatchFeature(
00065 |         data={
00066 |             "backbone_features": torch.randn(batch_size, 6, config.backbone_embedding_dim),
00067 |             "backbone_attention_mask": attention,
00068 |             "image_mask": image,
00069 |         }
00070 |     )

00073 | def _inputs(config, batch_size=2):
00074 |     shape = (batch_size, config.a2a_future_horizon, config.max_action_dim)
00075 |     continuous = torch.zeros(shape)
00076 |     continuous[..., :6] = 1
00077 |     auxiliary = torch.zeros(shape)
00078 |     auxiliary[..., 6:] = 1
00079 |     return BatchFeature(
00080 |         data={
00081 |             "history_action_canonical": torch.randn(shape),
00082 |             "future_action_canonical": torch.randn(shape).clamp(-1, 1),
00083 |             "history_action_mask": continuous.clone(),
00084 |             "future_action_mask": torch.ones(shape),
00085 |             "continuous_action_mask": continuous,
00086 |             "auxiliary_action_mask": auxiliary,
00087 |             "binary_action_mask": torch.zeros(shape),
00088 |             "categorical_action_mask": torch.zeros(shape),
00089 |             "categorical_group_index": torch.full(shape, -1, dtype=torch.long),
00090 |             "embodiment_id": torch.zeros(batch_size, dtype=torch.long),
00091 |             "state": torch.randn(batch_size, 1, config.max_state_dim),
00092 |         }
00093 |     )
