00016 | """Bitwise parity of the inference image path after removing the PIL round-trip.
00017 | 
00018 | ``_apply_vlm_processing`` used to convert every CHW frame to PIL before handing
00019 | it to the Qwen3VL processor; it now passes CHW uint8 tensors directly. These
00020 | tests assert the final model inputs are bitwise identical to a verbatim
00021 | reimplementation of the old PIL path, on the real checkpoint processor.
00022 | """

00028 | 
00029 |

00031 |

00035 | @pytest.fixture(scope="module")
00036 | def vlm_processor(load_hf_model_weights):
00037 |     """Real Qwen3VL processor; skips when the gated repo is unreachable.
00038 | 
00039 |     Only processor configs/tokenizer files are fetched (no model weights).
00040 |     Uses the ``load_hf_model_weights`` opt-in: the suite-wide
00041 |     ``GROOT_SKIP_HF_MODEL_WEIGHTS=1`` harness would otherwise stub the load and
00042 |     make the parity assertions vacuous.
00043 |     """
00044 |     from gr00t.model.gr00t_n1d7.processing_gr00t_n1d7 import build_processor

00053 | def _reference_pil_outputs(processor, frames_chw: np.ndarray, language: str):
00054 |     """Verbatim reimplementation of the pre-change path: CHW -> PIL -> processor."""
00055 |     pil_images = [Image.fromarray(np.transpose(v, (1, 2, 0))) for v in frames_chw]
00056 |     conversation = [
00057 |         {
00058 |             "role": "user",
00059 |             "content": [
00060 |                 *[{"type": "image", "image": img} for img in pil_images],
00061 |                 {"type": "text", "text": language},
00062 |             ],
00063 |         }
00064 |     ]
00065 |     text = processor.apply_chat_template(conversation, tokenize=False, add_generation_prompt=True)
00066 |     return processor(text=[text], images=pil_images, return_tensors="pt", padding=True)
00067 | 
00068 | 
00069 | def _direct_outputs(processor, frames_chw: np.ndarray, language: str):
00070 |     """The new path: CHW uint8 tensors handed to the processor directly."""
00071 |     frames = [torch.as_tensor(v) for v in frames_chw]
00072 |     conversation = [
00073 |         {
00074 |             "role": "user",
00075 |             "content": [
00076 |                 *[{"type": "image", "image": img} for img in frames],
00077 |                 {"type": "text", "text": language},
00078 |             ],
00079 |         }
00080 |     ]
00081 |     text = processor.apply_chat_template(conversation, tokenize=False, add_generation_prompt=True)
00082 |     return processor(text=[text], images=frames, return_tensors="pt", padding=True)
