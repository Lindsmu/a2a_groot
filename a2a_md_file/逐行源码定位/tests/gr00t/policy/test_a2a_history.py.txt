00011 | def _actual(value: float, batch_size: int = 2) -> dict[str, np.ndarray]:
00012 |     return {
00013 |         "arm": np.full((batch_size, 2), value, dtype=np.float32),
00014 |         "gripper": np.full((batch_size, 1, 1), value + 100, dtype=np.float32),
00015 |     }
00016 | 
00017 | 
00018 | def _external(start: float, batch_size: int = 2, horizon: int = 3) -> dict[str, np.ndarray]:
00019 |     steps = np.arange(start, start + horizon, dtype=np.float32)
00020 |     arm = np.broadcast_to(steps[None, :, None], (batch_size, horizon, 2)).copy()
00021 |     gripper = np.broadcast_to((steps + 100)[None, :, None], (batch_size, horizon, 1)).copy()
00022 |     return {"arm": arm, "gripper": gripper}
