00036 | def _trajectory_mask(mask: Tensor | None, reference: Tensor) -> Tensor:
00037 |     """Return a boolean mask broadcast to ``[batch, time, dimension]``."""
00038 |     if mask is None:
00039 |         return torch.ones_like(reference, dtype=torch.bool)
00040 |     if mask.ndim == reference.ndim - 1:
00041 |         mask = mask.unsqueeze(-1)
00042 |     try:
00043 |         return torch.broadcast_to(
00044 |             mask.to(device=reference.device, dtype=torch.bool), reference.shape

00057 |         return torch.ones(shape, dtype=torch.bool, device=reference.device)

00059 |         mask = mask.squeeze(-1)
00060 |     if mask.ndim != 2:
00061 |         raise ValueError(f"expected a [batch, sequence] mask, got shape {tuple(mask.shape)}")

00070 | def _masked_token_mean(hidden_states: Tensor, mask: Tensor) -> Tensor:
00071 |     weights = mask.to(dtype=hidden_states.dtype).unsqueeze(-1)
00072 |     masked_states = torch.where(mask.unsqueeze(-1), hidden_states, torch.zeros_like(hidden_states))
00073 |     total = masked_states.sum(dim=1)
00074 |     count = weights.sum(dim=1).clamp_min(1.0)
00075 |     return total / count

00078 | def masked_l1_loss(
00079 |     prediction: Tensor,
00080 |     target: Tensor,
00081 |     mask: Tensor | None = None,
00082 | ) -> Tensor:
00083 |     """Compute elementwise L1 averaged only over valid entries.
00084 | 
00085 |     An all-false mask produces a differentiable zero instead of a NaN.  Masks
00086 |     may omit trailing singleton dimensions as long as they are broadcastable to
00087 |     the prediction.
00088 |     """
00089 |     if prediction.shape != target.shape:
00090 |         raise ValueError(
00091 |             f"prediction and target must have the same shape, got "
00092 |             f"{tuple(prediction.shape)} and {tuple(target.shape)}"
00093 |         )
00094 |     if mask is None:
00095 |         return (prediction - target).abs().mean()
00096 |     while mask.ndim < prediction.ndim:
00097 |         mask = mask.unsqueeze(-1)
00098 |     try:
00099 |         valid = torch.broadcast_to(
00100 |             mask.to(device=prediction.device, dtype=torch.bool), prediction.shape
00101 |         )
00102 |     except RuntimeError as runtime_error:
00103 |         raise ValueError(
00104 |             f"mask with shape {tuple(mask.shape)} is not broadcastable to loss shape "
00105 |             f"{tuple(prediction.shape)}"
00106 |         ) from runtime_error
00107 |     difference = torch.where(valid, prediction - target, torch.zeros_like(prediction))
00108 |     weights = valid.to(dtype=prediction.dtype)
00109 |     return difference.abs().sum() / weights.sum().clamp_min(1.0)

00112 | class SinusoidalTimeEmbedding(nn.Module):
00113 |     """Continuous sinusoidal embedding for flow times in ``[0, 1]``."""
00114 | 
00115 |     def __init__(self, embedding_dim: int, max_period: float = 10_000.0):
00116 |         super().__init__()
00117 |         if embedding_dim <= 0:
00118 |             raise ValueError("embedding_dim must be positive")
00119 |         if max_period <= 0:
00120 |             raise ValueError("max_period must be positive")
00121 |         self.embedding_dim = embedding_dim
00122 |         self.max_period = max_period
00123 | 
00124 |     def forward(self, timesteps: Tensor) -> Tensor:
00125 |         if timesteps.ndim == 2 and timesteps.shape[-1] == 1:
00126 |             timesteps = timesteps.squeeze(-1)
00127 |         if timesteps.ndim != 1:
00128 |             raise ValueError(
00129 |                 f"expected one continuous timestep per batch item, got {tuple(timesteps.shape)}"
00130 |             )
00131 | 
00132 |         half_dim = self.embedding_dim // 2
00133 |         if half_dim == 0:
00134 |             return timesteps.float().unsqueeze(-1)
00135 |         exponent = (
00136 |             -math.log(self.max_period)
00137 |             * torch.arange(half_dim, device=timesteps.device, dtype=torch.float32)
00138 |             / max(half_dim, 1)
00139 |         )
00140 |         angles = timesteps.float().unsqueeze(-1) * exponent.exp().unsqueeze(0)
00141 |         embedding = torch.cat((angles.sin(), angles.cos()), dim=-1)
00142 |         if self.embedding_dim % 2:
00143 |             embedding = F.pad(embedding, (0, 1))
00144 |         return embedding

00147 | class ActionTrajectoryEncoder(nn.Module):
00148 |     """Encode a masked action/proprioception trajectory into one latent vector.
00149 | 
00150 |     The same encoder instance is intended to encode both the executed history
00151 |     ``z0`` and future demonstration trajectory ``z1``.  A category-specific
00152 |     input projection accommodates different embodiment semantics while the
00153 |     three temporal convolutions and latent projection are shared.
00154 |     """
00155 | 
00156 |     def __init__(
00157 |         self,
00158 |         action_dim: int,
00159 |         latent_dim: int = 512,
00160 |         num_embodiments: int = 1,
00161 |         conv_channels: tuple[int, int, int] = (128, 256, 512),
00162 |         kernel_size: int = 5,
00163 |     ):
00164 |         super().__init__()
00165 |         if action_dim <= 0 or latent_dim <= 0 or num_embodiments <= 0:
00166 |             raise ValueError("action_dim, latent_dim, and num_embodiments must be positive")
00167 |         if len(conv_channels) != 3 or any(channel <= 0 for channel in conv_channels):
00168 |             raise ValueError("conv_channels must contain exactly three positive channel sizes")
00169 |         if kernel_size <= 0 or kernel_size % 2 == 0:
00170 |             raise ValueError("kernel_size must be a positive odd integer")
00171 | 
00172 |         self.action_dim = action_dim
00173 |         self.latent_dim = latent_dim
00174 |         self.num_embodiments = num_embodiments
00175 |         self.input_adapter = CategorySpecificLinear(num_embodiments, action_dim, conv_channels[0])
00176 |         in_channels = (conv_channels[0], *conv_channels[:-1])
00177 |         padding = kernel_size // 2
00178 |         self.convolutions = nn.ModuleList(
00179 |             nn.Conv1d(input_channel, output_channel, kernel_size, padding=padding)
00180 |             for input_channel, output_channel in zip(in_channels, conv_channels, strict=True)
00181 |         )
00182 |         # Per-token normalization does not let synthetic/masked timesteps alter
00183 |         # the statistics of valid timesteps (unlike a temporal GroupNorm).
00184 |         self.normalizations = nn.ModuleList(nn.LayerNorm(channel) for channel in conv_channels)
00185 |         self.latent_projection = nn.Linear(conv_channels[-1], latent_dim)
00186 | 
00187 |     def forward(
00188 |         self,
00189 |         trajectory: Tensor,
00190 |         embodiment_ids: Tensor,
00191 |         trajectory_mask: Tensor | None = None,
00192 |     ) -> Tensor:
00193 |         if trajectory.ndim != 3 or trajectory.shape[-1] != self.action_dim:
00194 |             raise ValueError(
00195 |                 f"expected trajectory [batch, time, {self.action_dim}], got "
00196 |                 f"{tuple(trajectory.shape)}"
00197 |             )
00198 |         batch_size = trajectory.shape[0]
00199 |         if embodiment_ids.shape != (batch_size,):
00200 |             raise ValueError(
00201 |                 f"expected embodiment_ids shape ({batch_size},), got {tuple(embodiment_ids.shape)}"
00202 |             )
00203 | 
00204 |         element_mask = _trajectory_mask(trajectory_mask, trajectory)
00205 |         time_mask = element_mask.any(dim=-1)
00206 |         masked_trajectory = torch.where(element_mask, trajectory, torch.zeros_like(trajectory))
00207 | 
00208 |         hidden_states = self.input_adapter(masked_trajectory, embodiment_ids)
00209 |         time_weights = time_mask.to(dtype=hidden_states.dtype).unsqueeze(1)
00210 |         hidden_states = hidden_states.transpose(1, 2) * time_weights
00211 |         for convolution, normalization in zip(self.convolutions, self.normalizations, strict=True):
00212 |             hidden_states = convolution(hidden_states).transpose(1, 2)
00213 |             hidden_states = normalization(hidden_states).transpose(1, 2)
00214 |             hidden_states = F.silu(hidden_states) * time_weights
00215 | 
00216 |         pooled = hidden_states.sum(dim=-1) / time_weights.sum(dim=-1).clamp_min(1.0)
00217 |         latent = self.latent_projection(pooled)
00218 |         has_valid_token = time_mask.any(dim=-1, keepdim=True)
00219 |         return latent * has_valid_token.to(dtype=latent.dtype)

00588 | class TemporalSelfAttention(nn.Module):
00589 |     """ONNX-friendly multi-head self-attention over trajectory time tokens.
00590 | 
00591 |     ``nn.MultiheadAttention`` can dispatch to PyTorch's fused
00592 |     ``aten::_native_multi_head_attention`` during legacy ONNX tracing, which is
00593 |     not exportable in the deployment toolchain used by this repository.  The
00594 |     explicit QKV/MatMul/Softmax form below is mathematically equivalent and
00595 |     lowers to standard ONNX operators understood by ONNX Runtime and TensorRT.
00596 |     """
00597 | 
00598 |     def __init__(self, token_dim: int, num_heads: int, dropout: float = 0.0):
00599 |         super().__init__()
00600 |         if token_dim <= 0 or num_heads <= 0:
00601 |             raise ValueError("attention token_dim and num_heads must be positive")
00602 |         if token_dim % num_heads != 0:
00603 |             raise ValueError("attention token_dim must be divisible by num_heads")
00604 |         if not 0.0 <= dropout < 1.0:
00605 |             raise ValueError("attention dropout must be in [0, 1)")
00606 |         self.token_dim = token_dim
00607 |         self.num_heads = num_heads
00608 |         self.head_dim = token_dim // num_heads
00609 |         self.scale = self.head_dim**-0.5
00610 |         self.qkv_projection = nn.Linear(token_dim, 3 * token_dim)
00611 |         self.attention_dropout = nn.Dropout(dropout)
00612 |         self.output_projection = nn.Linear(token_dim, token_dim)
00613 |         self.output_dropout = nn.Dropout(dropout)
00614 | 
00615 |     def forward(self, hidden_states: Tensor) -> Tensor:
00616 |         if hidden_states.ndim != 3 or hidden_states.shape[-1] != self.token_dim:
00617 |             raise ValueError(
00618 |                 f"expected attention input [batch, tokens, {self.token_dim}], got "
00619 |                 f"{tuple(hidden_states.shape)}"
00620 |             )
00621 |         batch_size, num_tokens, _ = hidden_states.shape
00622 |         qkv = self.qkv_projection(hidden_states)
00623 |         qkv = qkv.reshape(batch_size, num_tokens, 3, self.num_heads, self.head_dim)
00624 |         qkv = qkv.permute(2, 0, 3, 1, 4)
00625 |         query, key, value = qkv.unbind(dim=0)
00626 |         attention_scores = torch.matmul(query, key.transpose(-2, -1)) * self.scale
00627 |         attention_probabilities = self.attention_dropout(attention_scores.softmax(dim=-1))

00629 |         attended = attended.transpose(1, 2).reshape(batch_size, num_tokens, self.token_dim)
