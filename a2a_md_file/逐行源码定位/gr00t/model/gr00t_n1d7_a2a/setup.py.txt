00115 |     def _configure_a2a_windows(self) -> None:
00116 |         history = int(self.model_config.a2a_history_horizon)
00117 |         future = int(self.model_config.a2a_future_horizon)
00118 |         if history != future:
00119 |             raise ValueError("Strict latent A2A requires equal history and future horizons")
00120 |         if int(self.model_config.action_horizon) != future:
00121 |             raise ValueError(
00122 |                 "action_horizon must equal a2a_future_horizon for the strict A2A variant"
00123 |             )
00124 |         if int(self.model_config.state_history_length) != 1:
00125 |             raise ValueError(
00126 |                 "state_history_length must remain 1; A2A history is carried separately"
00127 |             )
00128 | 
00129 |         updated = {}
00130 |         for embodiment, modalities in self.config.data.modality_configs.items():
00131 |             modalities = deepcopy(modalities)
00132 |             modalities["state"].delta_indices = list(range(-(history - 1), 1))
00133 |             modalities["action"].delta_indices = list(range(future))
00134 |             updated[embodiment] = modalities
00135 |         self.config.data.modality_configs = updated
