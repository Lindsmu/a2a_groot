00037 | def _masked_bce_with_logits(
00038 |     logits: torch.Tensor, target: torch.Tensor, mask: torch.Tensor
00039 | ) -> torch.Tensor:
00040 |     valid = mask.to(device=logits.device, dtype=torch.bool)
00041 |     safe_logits = torch.where(valid, logits, torch.zeros_like(logits))
00042 |     safe_target = torch.where(valid, target, torch.zeros_like(target))
00043 |     loss = F.binary_cross_entropy_with_logits(safe_logits, safe_target, reduction="none")
