00034 | # Convert tensors to lists for JSON serialization
00035 | def convert_tensors_to_lists(obj):
00036 |     """Recursively convert tensors to lists in nested dictionaries/lists."""
00037 |     if torch.is_tensor(obj) or isinstance(obj, np.ndarray):
00038 |         return obj.tolist()
00039 |     elif isinstance(obj, dict):
00040 |         return {key: convert_tensors_to_lists(value) for key, value in obj.items()}
00041 |     elif isinstance(obj, (list, tuple)):
00042 |         return [convert_tensors_to_lists(item) for item in obj]
00043 |     else:
00044 |         return obj
