00058 | ### Projector-index assignments, declared as ``{projector_index: {tags}}``.
00059 | #
00060 | # This grouped form is the source of truth: a tag only shares a projector with
00061 | # another tag if it is deliberately placed inside that index's set, so an
00062 | # accidental collision can't slip in unnoticed. Multiple tags share an index
00063 | # only when they describe the *same physical embodiment*. To add a brand-new
00064 | # embodiment, give it an unused index; to add a data-source/subtask variant of
00065 | # an existing one, add its tag to that group.
00066 | #
00067 | # ``EMBODIMENT_TAG_TO_PROJECTOR_INDEX`` below is derived from this and is the
00068 | # public, tag-keyed lookup used everywhere else.

00099 | def _build_tag_to_projector_index(groups: dict[int, set[str]]) -> dict[str, int]:
00100 |     """Flatten ``{index: {tags}}`` into ``{tag: index}``.
00101 | 
00102 |     Guards against a tag accidentally appearing in two groups, which would
00103 |     otherwise be silently resolved by insertion order into a single mapping.
00104 |     """
00105 |     mapping: dict[str, int] = {}
00106 |     for index, tags in groups.items():
00107 |         for tag in tags:
00108 |             if tag in mapping:
00109 |                 raise ValueError(
00110 |                     f"Embodiment tag {tag!r} is assigned to multiple projector "
00111 |                     f"indices ({mapping[tag]} and {index}) in "
00112 |                     "_PROJECTOR_INDEX_GROUPS; each tag must map to exactly one index."
00113 |                 )
00114 |             mapping[tag] = index
00115 |     return mapping

00132 | def validate_action_horizons(modality_configs, max_action_horizon: int) -> None:
00133 |     """Fail at processor construction if any configured embodiment's action horizon
00134 |     (the number of action ``delta_indices``) exceeds ``max_action_horizon``.
00135 | 
00136 |     ``max_action_horizon`` is set from the model's ``action_horizon``; without this
00137 |     check a horizon/model mismatch (e.g. a 50-step embodiment on a 40-step model)
00138 |     surfaces only deep in the first forward, after the model and dataset are built.
00139 |     """
00140 |     offenders: dict[str, int] = {}
00141 |     for tag, config in modality_configs.items():
00142 |         action = config.get("action") if isinstance(config, dict) else None
00143 |         delta_indices = getattr(action, "delta_indices", None)
00144 |         if delta_indices is None:
00145 |             continue
00146 |         horizon = len(delta_indices)
00147 |         if horizon > max_action_horizon:
00148 |             offenders[tag] = horizon
00149 |     if offenders:
00150 |         required = max(offenders.values())
00151 |         details = ", ".join(f"{tag}={horizon}" for tag, horizon in sorted(offenders.items()))
00152 |         raise ValueError(
00153 |             f"Embodiment action horizon exceeds max_action_horizon ({max_action_horizon}): "
00154 |             f"{details}. Increase model config action_horizon to >= {required} (or reduce the "
00155 |             "embodiment action delta_indices)."
00156 |         )
