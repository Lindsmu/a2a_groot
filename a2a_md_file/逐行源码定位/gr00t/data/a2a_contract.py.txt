00015 | def canonical_contract_sha256(contract: dict[str, Any]) -> str:
00016 |     """Return a deterministic SHA-256 over a contract excluding any stored hash."""
00017 |     payload = dict(contract)
00018 |     payload.pop("sha256", None)
00019 |     encoded = json.dumps(
00020 |         payload,
00021 |         ensure_ascii=False,
00022 |         sort_keys=True,
00023 |         separators=(",", ":"),
00024 |     ).encode("utf-8")
00025 |     return hashlib.sha256(encoded).hexdigest()

00066 |             )

00068 |         raise ValueError("A2A channel contract requires a non-empty embodiment")

00070 |         raise ValueError("A2A channel contract requires a non-empty channels list")

00082 |         "time_alignment",
00083 |     )

00086 |         for key in required_provenance
00087 |     ):

00089 |             "A2A channel contract provenance is incomplete; it must bind the dataset/schema, "
00090 |             "canonicalizer code, exporter, target definition, and time alignment"
00091 |         )

00093 |         "dataset_fingerprint_sha256",
00094 |         "source_schema_sha256",
00095 |         "canonicalizer_sha256",
00096 |     ):

00099 |             raise ValueError(f"A2A provenance {key} must be a lowercase SHA-256")

00110 |         raise ValueError(
00111 |             "A2A channel contract must explicitly bind controller type/version/scale/frame"
00112 |         )
00113 |     channels = normalized["channels"]
00114 |     expected_start = 0
00115 |     seen_action_keys = set()
00116 |     for channel in channels:
00117 |         if not isinstance(channel, dict):
00118 |             raise TypeError("Each A2A contract channel must be a JSON object")
00119 |         action_key = channel.get("action_key")
00120 |         if not isinstance(action_key, str) or not action_key or action_key in seen_action_keys:
00121 |             raise ValueError("A2A contract action keys must be unique non-empty strings")
00122 |         seen_action_keys.add(action_key)

00133 |         raise ValueError("A2A channel contract SHA-256 does not match its contents")
