00038 | def _rec_to_dtype(x: Any, dtype: torch.dtype) -> Any:
00039 |     """Recursively convert all floating point tensors in a nested structure to the given dtype.
00040 | 
00041 |     Args:
00042 |         x: Input data structure (tensor, dict, list, or other)
00043 |         dtype: Target torch dtype for floating point tensors
00044 | 
00045 |     Returns:
00046 |         Data structure with floating point tensors converted to target dtype
00047 | 
00048 |     Warning:
00049 |         Non-floating point tensors will be left as is.
00050 |     """
00051 |     if isinstance(x, torch.Tensor) and torch.is_floating_point(x):
00052 |         return x.to(dtype=dtype)
00053 |     # Handle dict-like objects (tianshou.BatchFeature is not dict but has items() method)
00054 |     elif isinstance(x, dict) or hasattr(x, "items"):
00055 |         return {k: _rec_to_dtype(v, dtype) for k, v in x.items()}  # type: ignore
00056 |     elif isinstance(x, list):
00057 |         return [_rec_to_dtype(v, dtype) for v in x]
00058 |     else:
00059 |         return x

00412 |         Pipeline:
00413 |         1. Unbatch observations into individual samples
00414 |         2. Convert each to VLAStepData and process
00415 |         3. Collate into model input batch

00417 |         5. Decode and unnormalize actions

00420 |             observation: Batched observation dictionary

00423 |         Returns:

00435 |         states = []
