00016 | """Custom Trainer with simple profiling utilities.
00017 | 
00018 | This subclass of HuggingFace's ``Trainer`` measures:
00019 | 1. Data loading latency (time between the end of the previous ``training_step`` and
00020 |    the start of the current ``training_step``).
00021 | 2. Forward-pass latency (time spent inside the base ``training_step`` implementation,
00022 |    which essentially wraps the model's forward / loss computation).
00023 | 
00024 | The statistics are logged via ``self.log`` every ``profile_log_interval`` steps and
00025 | also sent to the standard ``logging`` logger.  This is *not* meant to be a fully
00026 | fledged profiler – it is a quick, lightweight way to confirm whether the training
00027 | pipeline is bottlenecked by data loading or by the model's computation.
00028 | """

00054 |     def __init__(self, buffer, bs, collator, total_steps):

00068 |         if self._produced >= self._total_steps:
00069 |             raise StopIteration

00077 | class _PrefetchIterator:
00078 |     def __init__(self, buffer, bs, collate_fn, total_steps):
00079 |         self.buffer = buffer
00080 |         self.bs = bs
00081 |         self.collate = collate_fn
00082 |         self.total = total_steps
00083 |         self.produced = 0
00084 | 
00085 |         self._q = queue.Queue(maxsize=4)
00086 |         self._stop = False
00087 | 
00088 |         # Start background worker
00089 |         self._worker = threading.Thread(target=self._fill)
00090 |         self._worker.daemon = True
00091 |         self._worker.start()
00092 | 
00093 |     def _fill(self):
00094 |         while not self._stop:
00095 |             if self.produced + self._q.qsize() >= self.total:
00096 |                 break
00097 |             # block if queue is full
00098 |             samples = self.buffer.sample_batch(self.bs)
00099 |             batch = self.collate(samples)
00100 |             self._q.put(batch)
00101 | 
00102 |     def __iter__(self):
00103 |         return self
00104 | 
00105 |     def __len__(self):
00106 |         return self.total
00107 | 
00108 |     def __next__(self):
00109 |         if self.produced >= self.total:
00110 |             self._stop = True
00111 |             # in case worker is blocked on put()
00112 |             raise StopIteration
00113 |         batch = self._q.get()  # this will block until the next batch is ready
00114 |         self.produced += 1

00118 | def _batch_accuracy(
00119 |     preds: torch.Tensor, labels: torch.Tensor, action_offset: Optional[int] = None
00120 | ) -> torch.Tensor:  # noqa: D401
00121 |     """Compute token-level accuracy, ignoring ``-100`` label positions.
00122 | 
00123 |     Args:
00124 |         preds: Predicted token ids of shape ``(batch, seq_len)``.
00125 |         labels: Ground-truth label ids with the same shape as ``preds``.
00126 | 
00127 |     Returns:
00128 |         Scalar tensor with the fraction of correctly predicted labels in the
00129 |         current batch.
00130 |     """
00131 |     # casual prediction
00132 |     # Shift so that tokens < n predict n
00133 |     # https://github.com/huggingface/transformers/blob/main/src/transformers/loss/loss_utils.py#L60
00134 |     preds = preds[:, :-1]
00135 |     labels = labels[:, 1:]
00136 | 
00137 |     # Ignore positions with label == -100 (HF convention)
00138 |     mask = labels != -100
00139 | 
00140 |     if action_offset is not None:
00141 |         # we offset the labels to the action tokens range, with normal tokens in the negatives
00142 |         labels = labels - action_offset
00143 | 
00144 |     correct = (preds == labels) & mask
00145 | 
00146 |     # Avoid division by zero for empty masks (should not happen in practice)
00147 |     denom = mask.sum().clamp(min=1)
00148 |     accuracy = correct.sum().float() / denom.float()
00149 |     return accuracy

00152 | def _should_log_step_metrics(
00153 |     global_step: int,
00154 |     logging_steps: int | float | None,
00155 |     last_logged_step: int | None,
00156 | ) -> bool:
00157 |     """Return whether per-step metrics should be emitted exactly once.
00158 | 
00159 |     ``compute_loss`` may run several times at one optimizer step under gradient
00160 |     accumulation. Step zero is likewise visited once per micro-batch, so it is
00161 |     deliberately excluded.
00162 |     """
00163 |     if global_step <= 0 or global_step == last_logged_step:
00164 |         return False
00165 |     try:
00166 |         interval = int(logging_steps) if logging_steps is not None else 0
00167 |     except (TypeError, ValueError, OverflowError):
00168 |         return False
00169 |     return interval > 0 and global_step % interval == 0
