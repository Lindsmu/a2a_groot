00127 |     def validate_paper_architecture(self) -> None:
00128 |         """Reject drift from the repository's fixed paper-core migration profile.

00133 |         normalization details. Those are fixed repository implementation
00134 |         choices, not claims about undisclosed author code.
00135 |         """
00136 |         if self.a2a_flow_backbone not in {"mlp", "dit"}:
00137 |             raise ValueError(
00138 |                 f"a2a_flow_backbone must be 'mlp' or 'dit', got {self.a2a_flow_backbone!r}"
00139 |             )
00140 |         dit_integer_fields = {

00142 |             "a2a_dit_num_layers": self.a2a_dit_num_layers,
00143 |             "a2a_dit_num_heads": self.a2a_dit_num_heads,
00144 |             "a2a_dit_mlp_ratio": self.a2a_dit_mlp_ratio,
00145 |         }
00146 |         nonpositive = {name: value for name, value in dit_integer_fields.items() if value <= 0}

00152 |             raise ValueError("a2a_dit_dropout must be in [0, 1)")
