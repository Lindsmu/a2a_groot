00032 | def _build_safe_tree(obj: Any) -> Any:
00033 |     """Convert an ``asdict()`` tree into one that ``yaml.safe_dump`` accepts.
00034 | 
00035 |     ``yaml.safe_dump`` only knows how to write plain primitives (str / int /
00036 |     bool / None / list / dict); it raises ``RepresenterError`` on anything
00037 |     else. The only "anything else" we hit today is ``Enum`` values —
00038 |     ``MODALITY_CONFIGS`` carries ``ActionConfig`` fields holding
00039 |     ``ActionRepresentation`` / ``ActionType`` / ``ActionFormat`` enums — so we
00040 |     walk the tree once and replace each ``Enum`` with its ``.value`` string
00041 |     (e.g. ``ActionRepresentation.RELATIVE`` -> ``"relative"``). An explicit
00042 |     walk is easier to audit than a custom ``SafeDumper`` and leaves no global
00043 |     YAML state behind.
00044 |     """
00045 |     if isinstance(obj, Enum):
00046 |         return obj.value
00047 |     if isinstance(obj, dict):
00048 |         return {k: _build_safe_tree(v) for k, v in obj.items()}
00049 |     if isinstance(obj, (list, tuple)):
00050 |         return [_build_safe_tree(v) for v in obj]
00051 |     return obj

00054 | def _load_safe_yaml(path: Path):
00055 |     """Load YAML with ``yaml.safe_load`` (never the object-constructing
00056 |     ``yaml.Loader``), turning the safe loader's rejection of legacy
00057 |     ``!!python/object`` tags into a friendly migration error.
00058 | 
00059 |     ``yaml.safe_load`` refuses those tags with a ``ConstructorError``; we
00060 |     re-raise it as a ``ValueError`` that explains the old config can't load
00061 |     and needs a one-time re-save, instead of leaking a cryptic PyYAML
00062 |     traceback.
00063 |     """
00064 |     text = path.read_text()
00065 |     try:
00066 |         return yaml.safe_load(text)
00067 |     except yaml.constructor.ConstructorError as e:
00068 |         raise ValueError(
00069 |             f"{path}: rejected unsafe legacy config YAML (contains "
00070 |             f"{e.problem!r}). The pre-2026-05 Config.save() emitted "
00071 |             "PyYAML !!python/object tags which the loader is no longer "
00072 |             "willing to instantiate — that path was an arbitrary-code-"
00073 |             "execution gadget. Re-save the config via the current "
00074 |             "Config.save() to migrate to the plain-dict YAML format."
00075 |         ) from e

00090 |     def save(self, path: Path):
00091 |         """Save the config as plain key/value YAML (no Python-object tags).
00092 | 
00093 |         Writes ``asdict(self)`` with ``yaml.safe_dump``, so the file is human
00094 |         readable, diffable, and — crucially — builds no Python objects when
00095 |         read back. ``_build_safe_tree`` first lowers any ``Enum`` fields to
00096 |         plain strings, which ``yaml.safe_dump`` would otherwise refuse to
00097 |         write. Replaces the old ``yaml.dump(self)``, which embedded
00098 |         ``!!python/object`` tags that made loading a config able to run code.
00099 |         """
00100 |         path = Path(path)
00101 |         path.parent.mkdir(parents=True, exist_ok=True)
00102 | 
00103 |         with open(path, "w") as f:
00104 |             yaml.safe_dump(_build_safe_tree(asdict(self)), f, sort_keys=False)

00106 |     def load(self, path: Path):
00107 |         """Load config from a plain key/value YAML file into ``self``.
00108 | 
00109 |         Reads via the safe loader (so a malicious or legacy
00110 |         ``!!python/object`` file is refused with a migration error, never
00111 |         executed) and rebuilds the nested dataclasses with :meth:`load_dict`.
00112 |         """
00113 |         data = _load_safe_yaml(path)
00114 |         if not isinstance(data, dict):
00115 |             raise ValueError(
00116 |                 f"Invalid config file: {path}. Expected a YAML mapping at "
00117 |                 f"the top level (saved by Config.save()); got {type(data).__name__}."
00118 |             )
00119 |         self.load_dict(data)
00120 |         return self

00126 |                 raise ValueError("Config.model must be a mapping")

00131 |                 model_class = MODEL_CONFIG_TYPES.get(model_type)

00133 |                     raise ValueError(
00134 |                         f"Unknown model_type {model_type!r}; available types are "
00135 |                         f"{sorted(MODEL_CONFIG_TYPES)}"

00137 |             self.model = model_class(**model_data)
00138 |         if "data" in data:
00139 |             self.data = DataConfig(**data["data"])
00140 |             # Ensure nested datasets are converted to dataclass instances
